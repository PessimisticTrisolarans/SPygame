        function placeMines(safeRow, safeCol) {
            var rows = game.rows, cols = game.cols;
            for (var r = 0; r < rows; r++) {
                for (var c = 0; c < cols; c++) {
                    game.board[r][c] = 0;
                }
            }
            var safeZone = new Set();
            for (var dr = -1; dr <= 1; dr++) {
                for (var dc = -1; dc <= 1; dc++) {
                    var nr = safeRow + dr, nc = safeCol + dc;
                    if (nr >= 0 && nr < rows && nc >= 0 && nc < cols) safeZone.add(nr + ',' + nc);
                }
            }
            var candidates = [];
            for (var r = 0; r < rows; r++) {
                for (var c = 0; c < cols; c++) {
                    if (!safeZone.has(r + ',' + c)) candidates.push([r, c]);
                }
            }
            candidates.sort(function() { return Math.random() - 0.5; });
            var mineCount = Math.min(game.mines, candidates.length);
            for (var i = 0; i < mineCount; i++) {
                var pos = candidates[i];
                game.board[pos[0]][pos[1]] = -1;
            }
            for (var r = 0; r < rows; r++) {
                for (var c = 0; c < cols; c++) {
                    if (game.board[r][c] === -1) continue;
                    var n = 0;
                    for (var dr = -1; dr <= 1; dr++) {
                        for (var dc = -1; dc <= 1; dc++) {
                            var nr = r + dr, nc = c + dc;
                            if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && game.board[nr][nc] === -1) n++;
                        }
                    }
                    game.board[r][c] = n;
                }
            }
        }

        function drawCellAt(row, col, x, y, board, revealed, flagged, flashMine) {
            if (flashMine === undefined) flashMine = false;
            var cellSize = game.cellSize;
            if (revealed[row][col]) {
                if (flashMine) {
                    var flash = Math.floor(performance.now() / 300) % 2 === 0;
                    ctx.fillStyle = flash ? '#ff0000' : '#ffff00';
                } else {
                    ctx.fillStyle = '#3a3f5c';
                }
                ctx.fillRect(x, y, cellSize, cellSize);
                ctx.strokeStyle = '#1e1e2e';
                ctx.lineWidth = 0.5;
                ctx.strokeRect(x, y, cellSize, cellSize);

                if (board[row][col] === -1) {
                    if (mineImg.complete && mineImg.naturalWidth > 0) {
                        ctx.drawImage(mineImg, x, y, cellSize, cellSize);
                    } else {
                        var cx = x + cellSize / 2, cy = y + cellSize / 2, radius = cellSize * 0.3;
                        ctx.fillStyle = '#ff4757';
                        ctx.beginPath();
                        ctx.arc(cx, cy, radius, 0, Math.PI * 2);
                        ctx.fill();
                    }

                    if (flagged[row][col] === 1) {
                        if (fmImg.complete && fmImg.naturalWidth > 0) {
                            ctx.drawImage(fmImg, x, y, cellSize, cellSize);
                        } else {
                            var cx = x + cellSize / 2, cy = y + cellSize / 2, flagSize = cellSize * 0.3;
                            ctx.fillStyle = '#ff6b6b';
                            ctx.beginPath();
                            ctx.moveTo(cx - flagSize, cy - flagSize);
                            ctx.lineTo(cx + flagSize, cy - flagSize * 0.4);
                            ctx.lineTo(cx - flagSize, cy + flagSize * 0.4);
                            ctx.closePath();
                            ctx.fill();
                            ctx.strokeStyle = '#fff';
                            ctx.lineWidth = 2;
                            ctx.beginPath();
                            ctx.moveTo(cx - flagSize, cy - flagSize);
                            ctx.lineTo(cx - flagSize, cy + flagSize);
                            ctx.stroke();
                        }
                    }
                } else if (board[row][col] > 0) {
                    ctx.fillStyle = NUMBER_COLORS[board[row][col]];
                    ctx.font = 'bold ' + Math.floor(cellSize * 0.62) + 'px Arial';
                    ctx.textAlign = 'center';
                    ctx.textBaseline = 'middle';
                    ctx.fillText(board[row][col], x + cellSize / 2, y + cellSize / 2);
                }
            } else {
                ctx.fillStyle = '#4a4f6a';
                ctx.fillRect(x, y, cellSize, cellSize);
                ctx.strokeStyle = '#1e1e2e';
                ctx.lineWidth = 0.5;
                ctx.strokeRect(x, y, cellSize, cellSize);
                if (flagged[row][col] === 1) {
                    var cx = x + cellSize / 2, cy = y + cellSize / 2, flagSize = cellSize * 0.3;
                    ctx.fillStyle = '#ff6b6b';
                    ctx.beginPath();
                    ctx.moveTo(cx - flagSize, cy - flagSize);
                    ctx.lineTo(cx + flagSize, cy - flagSize * 0.4);
                    ctx.lineTo(cx - flagSize, cy + flagSize * 0.4);
                    ctx.closePath();
                    ctx.fill();
                    ctx.strokeStyle = '#fff';
                    ctx.lineWidth = 2;
                    ctx.beginPath();
                    ctx.moveTo(cx - flagSize, cy - flagSize);
                    ctx.lineTo(cx - flagSize, cy + flagSize);
                    ctx.stroke();
                } else if (flagged[row][col] === 2) {
                    ctx.fillStyle = '#fff';
                    ctx.font = 'bold ' + Math.floor(cellSize * 0.6) + 'px Arial';
                    ctx.textAlign = 'center';
                    ctx.textBaseline = 'middle';
                    ctx.fillText('?', x + cellSize / 2, y + cellSize / 2);
                }
            }
        }

        function revealCell(row, col) {
            if (!game || game.gameOver || game.win || game.isScrolling) return;
            if (row < 0 || row >= game.rows || col < 0 || col >= game.cols) return;
            if (game.revealed[row][col] || game.flagged[row][col] === 1) return;
            if (game.firstClick) {
                placeMines(row, col);
                game.firstClick = false;
                startTimer();
            }

            if (!game.endless && game.board[row][col] === -1 && isTwoChoiceProtection(row, col)) {
                handleTwoChoiceWin(row, col);
                return;
            }

            game.revealed[row][col] = true;
            game.score++;
            if (game.board[row][col] === -1) {
                if (game.endless) {
                    endlessHp -= 5;
                    updateHpDisplay();
                    if (endlessHp <= 0) {
                        endlessHp = 0;
                        game.gameOver = true;
                        smileyBtn.textContent = '💀';
                        var survival = formatTime(Math.floor((performance.now() - endlessStartTime) / 1000));
                        showOverlay('生命值耗尽！\n存活时间：' + survival);
                    }
                    updateMinesCount();
                    return;
                } else {
                    game.flashingMine = { row: row, col: col };
                    game.gameOver = true;
                    smileyBtn.textContent = '😵';
                    stopTimer();
                    for (var r = 0; r < game.rows; r++) {
                        for (var c = 0; c < game.cols; c++) {
                            if (game.board[r][c] === -1) game.revealed[r][c] = true;
                        }
                    }
                    showOverlay('踩到地雷！');
                    return;
                }
            }
            if (game.board[row][col] === 0) {
                var stack = [[row, col]];
                while (stack.length) {
                    var pos = stack.pop();
                    for (var dr = -1; dr <= 1; dr++) {
                        for (var dc = -1; dc <= 1; dc++) {
                            if (dr === 0 && dc === 0) continue;
                            var nr = pos[0] + dr, nc = pos[1] + dc;
                            if (nr >= 0 && nr < game.rows && nc >= 0 && nc < game.cols && !game.revealed[nr][nc] && game.flagged[nr][nc] !== 1) {
                                game.revealed[nr][nc] = true;
                                game.score++;
                                if (game.board[nr][nc] === 0) stack.push([nr, nc]);
                            }
                        }
                    }
                }
            }
            updateMinesCount();
            checkWin();
            if (autoFlagEnabled && !game.gameOver && !game.win) autoFlagAllMatchingNumbers();
        }

        function handleTwoChoiceWin(row, col) {
            var otherRow = -1, otherCol = -1;
            for (var r = 0; r < game.rows; r++) {
                for (var c = 0; c < game.cols; c++) {
                    if (!game.revealed[r][c] && !(r === row && c === col)) {
                        otherRow = r;
                        otherCol = c;
                    }
                }
            }
            if (otherRow === -1) return;
            game.board[row][col] = 0;
            game.board[otherRow][otherCol] = -1;
            var count = 0;
            for (var dr = -1; dr <= 1; dr++) {
                for (var dc = -1; dc <= 1; dc++) {
                    var nr = row + dr, nc = col + dc;
                    if (nr >= 0 && nr < game.rows && nc >= 0 && nc < game.cols && game.board[nr][nc] === -1) count++;
                }
            }
            game.board[row][col] = count;
            game.flagged[otherRow][otherCol] = 1;
            game.revealed[row][col] = true;
            updateMinesCount();
            game.win = true;
            smileyBtn.textContent = '😎';
            stopTimer();
            if (game.scrollTimer) clearInterval(game.scrollTimer);
            if (initialCellSizeForWin && game.cellSize !== initialCellSizeForWin) {
                setCellSize(initialCellSizeForWin, false);
                canvasWrapper.scrollTop = 0;
                canvasWrapper.scrollLeft = 0;
            }
            showOverlay('你赢了！（二选一保护）');
        }

        function isTwoChoiceProtection(row, col) {
            if (!game || game.gameOver || game.win) return false;
            if (game.board[row][col] !== -1 || game.revealed[row][col] || game.flagged[row][col] === 1) return false;
            var unrevealedCount = 0, unrevealedMineCount = 0;
            for (var r = 0; r < game.rows; r++) {
                for (var c = 0; c < game.cols; c++) {
                    if (!game.revealed[r][c]) {
                        unrevealedCount++;
                        if (game.board[r][c] === -1) unrevealedMineCount++;
                    }
                }
            }
            return unrevealedCount === 2 && unrevealedMineCount === 1;
        }

        function autoFlagAllMatchingNumbers() {
            if (!game || game.gameOver || game.win) return;
            for (var r = 0; r < game.rows; r++) {
                for (var c = 0; c < game.cols; c++) {
                    if (!game.revealed[r][c] || game.board[r][c] <= 0) continue;
                    var allUnrevealed = [];
                    var unflaggedUnrevealed = [];
                    for (var dr = -1; dr <= 1; dr++) {
                        for (var dc = -1; dc <= 1; dc++) {
                            if (dr === 0 && dc === 0) continue;
                            var nr = r + dr, nc = c + dc;
                            if (nr >= 0 && nr < game.rows && nc >= 0 && nc < game.cols && !game.revealed[nr][nc]) {
                                allUnrevealed.push([nr, nc]);
                                if (game.flagged[nr][nc] === 0) unflaggedUnrevealed.push([nr, nc]);
                            }
                        }
                    }
                    if (allUnrevealed.length === game.board[r][c]) {
                        for (var i = 0; i < unflaggedUnrevealed.length; i++) {
                            var pos = unflaggedUnrevealed[i];
                            game.flagged[pos[0]][pos[1]] = 1;
                        }
                    }
                }
            }
            updateMinesCount();
            draw();
        }

        function toggleFlag(r, c) {
            if (!game || game.gameOver || game.win || game.isScrolling) return;
            if (r < 0 || r >= game.rows || c < 0 || c >= game.cols) return;
            if (!game.endless && game.revealed[r][c]) return;
            if (game.endless && game.revealed[r][c] && game.board[r][c] !== -1) return;

            var current = game.flagged[r][c];
            var newFlag = 0;
            if (current === 0) newFlag = 1;
            else if (current === 1) newFlag = 2;
            else newFlag = 0;

            if (game.endless && current === 0 && newFlag === 1 && !game.revealed[r][c] && game.board[r][c] !== -1) {
                endlessWrongFlagCount++;
                if (endlessWrongFlagCount >= 5) {
                    var deduct = Math.floor(endlessWrongFlagCount / 5) * 3;
                    endlessHp -= deduct;
                    endlessWrongFlagCount = endlessWrongFlagCount % 5;
                    updateHpDisplay();
                    if (endlessHp <= 0) {
                        endlessHp = 0;
                        game.gameOver = true;
                        smileyBtn.textContent = '💀';
                        var survival = formatTime(Math.floor((performance.now() - endlessStartTime) / 1000));
                        showOverlay('生命值耗尽！\n存活时间：' + survival);
                        updateEndlessStats();
                        return;
                    }
                }
            }
            game.flagged[r][c] = newFlag;
            updateMinesCount();
            updateEndlessStats();
            draw();
            if (autoFlagEnabled && !game.gameOver && !game.win) autoFlagAllMatchingNumbers();
        }

        function chordCell(r, c) {
            if (!game || game.gameOver || game.win || game.isScrolling) return;
            if (r < 0 || r >= game.rows || c < 0 || c >= game.cols || !game.revealed[r][c] || game.board[r][c] <= 0) return;
            var flaggedCount = 0;
            var unflaggedUnrevealed = [];
            for (var dr = -1; dr <= 1; dr++) {
                for (var dc = -1; dc <= 1; dc++) {
                    if (dr === 0 && dc === 0) continue;
                    var nr = r + dr, nc = c + dc;
                    if (nr >= 0 && nr < game.rows && nc >= 0 && nc < game.cols) {
                        if (game.flagged[nr][nc] === 1) {
                            flaggedCount++;
                        } else if (!game.revealed[nr][nc] && game.flagged[nr][nc] === 0) {
                            unflaggedUnrevealed.push([nr, nc]);
                        }
                    }
                }
            }
            if (flaggedCount >= game.board[r][c]) {
                for (var i = 0; i < unflaggedUnrevealed.length; i++) {
                    revealCell(unflaggedUnrevealed[i][0], unflaggedUnrevealed[i][1]);
                }
            } else {
                game.flashCells = unflaggedUnrevealed;
                game.flashStartTime = performance.now();
            }
        }

        function checkWin() {
            if (game.endless) return;
            for (var r = 0; r < game.rows; r++) {
                for (var c = 0; c < game.cols; c++) {
                    if (game.board[r][c] !== -1 && !game.revealed[r][c]) return;
                }
            }
            game.win = true;
            handleWin();
        }

        function handleWin() {
            for (var r = 0; r < game.rows; r++) {
                for (var c = 0; c < game.cols; c++) {
                    if (!game.revealed[r][c] && game.flagged[r][c] === 0) game.flagged[r][c] = 1;
                }
            }
            updateMinesCount();
            smileyBtn.textContent = '😎';
            stopTimer();
            if (game.scrollTimer) clearInterval(game.scrollTimer);
            if (initialCellSizeForWin && game.cellSize !== initialCellSizeForWin) {
                setCellSize(initialCellSizeForWin, false);
                canvasWrapper.scrollTop = 0;
                canvasWrapper.scrollLeft = 0;
            }
            showOverlay('你赢了！');
        }

        function updateMinesCount() {
            if (!game) return;
            var totalMines = game.mines;
            if (game.endless) {
                totalMines = 0;
                for (var r = 0; r < game.rows; r++) {
                    for (var c = 0; c < game.cols; c++) {
                        if (game.board[r][c] === -1) totalMines++;
                    }
                }
            }
            var flags = game.flagged.flat().filter(function(f) { return f === 1; }).length;
            minesCountSpan.textContent = (totalMines - flags).toString().padStart(3, '0');
        }

        function updateHpDisplay() {
            hpDisplay.textContent = '❤️' + endlessHp;
            hpDisplay.style.color = '#ff0000';
            setTimeout(function() { hpDisplay.style.color = '#ff6b6b'; }, 300);
        }

        function updateEndlessStats() {
            if (!game || !game.endless) return;
            endlessStats.textContent = '错旗: ' + endlessWrongFlagCount + '/5\n未标雷: ' + endlessUnmarkedMineCount + '/3';
        }

        function formatTime(seconds) {
            var mins = Math.floor(seconds / 60);
            var secs = seconds % 60;
            return mins.toString().padStart(2, '0') + ':' + secs.toString().padStart(2, '0');
        }

        function performScroll() {
            if (!game || !game.endless || game.gameOver || game.win || game.isScrolling) return;
            game.isScrolling = true;
            var scrollRows = 5;
            var scrollDistance = scrollRows * game.cellSize;
            var duration = 400;
            var startTime = performance.now();
            var startBoard = game.board.map(function(row) { return row.slice(); });
            var startRevealed = game.revealed.map(function(row) { return row.slice(); });
            var startFlagged = game.flagged.map(function(row) { return row.slice(); });

            function animateScroll(now) {
                var elapsed = now - startTime;
                var progress = Math.min(elapsed / duration, 1);
                var offset = Math.floor(scrollDistance * progress);
                ctx.clearRect(0, 0, canvas.width, canvas.height);
                var cellSize = game.cellSize;
                for (var r = 0; r < game.rows; r++) {
                    for (var c = 0; c < game.cols; c++) {
                        var x = c * cellSize;
                        var y = r * cellSize - offset;
                        if (y + cellSize < 0) continue;
                        drawCellAt(r, c, x, y, startBoard, startRevealed, startFlagged);
                    }
                }
                if (progress < 1) {
                    requestAnimationFrame(animateScroll);
                } else {
                    completeScroll();
                }
            }
            requestAnimationFrame(animateScroll);
        }

        function completeScroll() {
            var unmarkedMines = 0;
            for (var r = 0; r < 5; r++) {
                for (var c = 0; c < game.cols; c++) {
                    if (game.board[r][c] === -1 && game.flagged[r][c] !== 1) unmarkedMines++;
                }
            }
            if (unmarkedMines > 0) {
                endlessUnmarkedMineCount += unmarkedMines;
                while (endlessUnmarkedMineCount >= 3) {
                    endlessUnmarkedMineCount -= 3;
                    endlessHp -= 2;
                    updateHpDisplay();
                    if (endlessHp <= 0) {
                        endlessHp = 0;
                        game.gameOver = true;
                        smileyBtn.textContent = '💀';
                        var survival = formatTime(Math.floor((performance.now() - endlessStartTime) / 1000));
                        showOverlay('生命值耗尽！\n存活时间：' + survival);
                        game.isScrolling = false;
                        return;
                    }
                }
            }
            for (var i = 0; i < 5; i++) {
                game.board.shift();
                game.revealed.shift();
                game.flagged.shift();
            }
            var mineProb = game.mines / (game.rows * game.cols);
            for (var i = 0; i < 5; i++) {
                var br = [], rr = [], fr = [];
                for (var c = 0; c < game.cols; c++) {
                    br.push(Math.random() < mineProb ? -1 : 0);
                    rr.push(false);
                    fr.push(0);
                }
                game.board.push(br);
                game.revealed.push(rr);
                game.flagged.push(fr);
            }
            for (var r = 0; r < game.rows; r++) {
                for (var c = 0; c < game.cols; c++) {
                    if (game.board[r][c] === -1) continue;
                    var n = 0;
                    for (var dr = -1; dr <= 1; dr++) {
                        for (var dc = -1; dc <= 1; dc++) {
                            var nr = r + dr, nc = c + dc;
                            if (nr >= 0 && nr < game.rows && nc >= 0 && nc < game.cols && game.board[nr][nc] === -1) n++;
                        }
                    }
                    game.board[r][c] = n;
                }
            }
            if (game.flashingMine && game.flashingMine.row < 5) game.flashingMine = null;
            updateMinesCount();
            updateEndlessStats();
            game.isScrolling = false;
            endlessPhase = 'loading';
            endlessPhaseStartTime = performance.now();
            rainbowBar.style.width = '0%';
            draw();
            if (autoFlagEnabled) autoFlagAllMatchingNumbers();
        }

        function updateRainbowBar() {
            if (!game || !game.endless || game.gameOver || game.win || game.isScrolling) return;
            var now = performance.now();
            var progress = 0;
            if (endlessPhase === 'countdown') {
                var elapsed = now - endlessPhaseStartTime;
                progress = Math.max(0, 1 - elapsed / ENDLESS_COUNTDOWN_MS);
                if (elapsed >= ENDLESS_COUNTDOWN_MS) {
                    endlessPhase = 'loading';
                    endlessPhaseStartTime = now;
                    progress = 0;
                }
            } else {
                var elapsed = now - endlessPhaseStartTime;
                progress = Math.min(1, elapsed / ENDLESS_LOADING_MS);
                if (progress >= 1) {
                    performScroll();
                    return;
                }
            }
            rainbowBar.style.width = (progress * 100) + '%';
        }

        function showOverlay(text) {
            resultText.textContent = text;
            overlay.style.display = 'flex';
        }

        function updateStatusBar() { }

        function draw() {
            if (!game) return;
            var rows = game.rows, cols = game.cols, cellSize = game.cellSize;
            var board = game.board, revealed = game.revealed, flagged = game.flagged;
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            var isFlashing = performance.now() - game.flashStartTime < FLASH_DURATION_MS;
            var flashSet = isFlashing ? new Set(game.flashCells.map(function(pos) { return pos[0] + ',' + pos[1]; })) : null;
            for (var r = 0; r < rows; r++) {
                for (var c = 0; c < cols; c++) {
                    var x = c * cellSize, y = r * cellSize;
                    if (flashSet && flashSet.has(r + ',' + c) && !revealed[r][c] && flagged[r][c] === 0) {
                        ctx.fillStyle = '#ffd43b';
                        ctx.fillRect(x, y, cellSize, cellSize);
                        ctx.strokeStyle = '#1e1e2e';
                        ctx.lineWidth = 0.5;
                        ctx.strokeRect(x, y, cellSize, cellSize);
                        continue;
                    }
                    var isFlashMine = game.flashingMine && r === game.flashingMine.row && c === game.flashingMine.col && revealed[r][c] && board[r][c] === -1;
                    drawCellAt(r, c, x, y, board, revealed, flagged, isFlashMine);
                }
            }
        }

        function gameLoop() {
            if (!game) return;
            if (!game.isScrolling) {
                draw();
                updateStatusBar();
                if (game.endless && !game.gameOver && !game.win) updateRainbowBar();
            }
            game.animationId = requestAnimationFrame(gameLoop);
        }

        function setCellSize(newSize, keepScroll) {
            if (keepScroll === undefined) keepScroll = true;
            if (!game) return;
            newSize = Math.max(minCellSize, Math.min(maxCellSize, newSize));
            if (newSize === game.cellSize) return;
            var oldWidth = canvas.width, oldHeight = canvas.height;
            var scrollRatioX = keepScroll ? canvasWrapper.scrollLeft / oldWidth : 0;
            var scrollRatioY = keepScroll ? canvasWrapper.scrollTop / oldHeight : 0;
            game.cellSize = newSize;
            canvas.width = game.cols * newSize;
            canvas.height = game.rows * newSize;
            canvas.style.width = canvas.width + 'px';
            canvas.style.height = canvas.height + 'px';
            if (keepScroll) {
                canvasWrapper.scrollLeft = scrollRatioX * canvas.width;
                canvasWrapper.scrollTop = scrollRatioY * canvas.height;
            } else {
                canvasWrapper.scrollLeft = 0;
                canvasWrapper.scrollTop = 0;
            }
            draw();
        }

        function getGridPosition(e) {
            var rect = canvas.getBoundingClientRect();
            var scaleX = canvas.width / rect.width, scaleY = canvas.height / rect.height;
            var clientX, clientY;
            if (e.touches && e.touches.length > 0) {
                clientX = e.touches[0].clientX;
                clientY = e.touches[0].clientY;
            } else {
                clientX = e.clientX;
                clientY = e.clientY;
            }
            var mx = (clientX - rect.left) * scaleX;
            var my = (clientY - rect.top) * scaleY;
            return { row: Math.floor(my / game.cellSize), col: Math.floor(mx / game.cellSize) };
        }

        function onPointerDown(e) {
            if (isPinching || (game && game.isScrolling)) return;
            if (!game || game.gameOver || game.win) return;
            if (e.button === 2) {
                e.preventDefault();
                var pos = getGridPosition(e);
                if (pos.row < 0 || pos.row >= game.rows || pos.col < 0 || pos.col >= game.cols) return;
                if (game.revealed[pos.row][pos.col] && game.board[pos.row][pos.col] > 0) chordCell(pos.row, pos.col);
                else toggleFlag(pos.row, pos.col);
                return;
            }
            e.preventDefault();
            var pos = getGridPosition(e);
            if (pos.row < 0 || pos.row >= game.rows || pos.col < 0 || pos.col >= game.cols) return;
            var clientX = e.touches ? e.touches[0].clientX : e.clientX;
            var clientY = e.touches ? e.touches[0].clientY : e.clientY;
            pressStartClientX = clientX;
            pressStartClientY = clientY;
            pressStartPos = pos;
            longPressTriggered = false;
            isDragging = false;
            longPressTimer = setTimeout(function() {
                if (pressStartPos) {
                    var pr = pressStartPos.row, pc = pressStartPos.col;
                    if (!game.revealed[pr][pc]) revealCell(pr, pc);
                    longPressTriggered = true;
                }
            }, LONG_PRESS_MS);
        }

        function onPointerMove(e) {
            if (isPinching || (game && game.isScrolling)) return;
            if (!game || game.gameOver || game.win || !pressStartPos) return;
            var clientX = e.touches ? e.touches[0].clientX : e.clientX;
            var clientY = e.touches ? e.touches[0].clientY : e.clientY;
            var dx = Math.abs(clientX - pressStartClientX), dy = Math.abs(clientY - pressStartClientY);
            if (dx > DRAG_THRESHOLD || dy > DRAG_THRESHOLD) {
                isDragging = true;
                if (longPressTimer) { clearTimeout(longPressTimer); longPressTimer = null; }
                longPressTriggered = false;
            }
        }

        function onPointerUp(e) {
            if (isPinching || (game && game.isScrolling)) return;
            if (!game || game.gameOver || game.win) return;
            if (e.button === 2) return;
            e.preventDefault();
            if (longPressTimer) { clearTimeout(longPressTimer); longPressTimer = null; }
            if (isDragging) { pressStartPos = null; isDragging = false; return; }
            if (longPressTriggered) { pressStartPos = null; longPressTriggered = false; return; }
            if (pressStartPos) {
                var pr = pressStartPos.row, pc = pressStartPos.col;
                if (game.firstClick) {
                    revealCell(pr, pc);
                } else if (game.endless && game.revealed[pr][pc] && game.board[pr][pc] === -1) {
                    toggleFlag(pr, pc);
                } else if (game.revealed[pr][pc] && game.board[pr][pc] > 0) {
                    chordCell(pr, pc);
                } else if (!game.revealed[pr][pc]) {
                    toggleFlag(pr, pc);
                }
            }
            pressStartPos = null;
            longPressTriggered = false;
        }

        function onPointerLeave() {
            if (longPressTimer) { clearTimeout(longPressTimer); longPressTimer = null; }
            pressStartPos = null;
            longPressTriggered = false;
            isDragging = false;
        }

        function getDistance(t1, t2) {
            var dx = t1.clientX - t2.clientX;
            var dy = t1.clientY - t2.clientY;
            return Math.sqrt(dx * dx + dy * dy);
        }

        canvas.addEventListener('touchstart', function(e) {
            if (e.touches.length === 2) {
                e.preventDefault();
                isPinching = true;
                pinchStartDistance = getDistance(e.touches[0], e.touches[1]);
                pinchStartCellSize = game ? game.cellSize : maxCellSize;
                if (longPressTimer) { clearTimeout(longPressTimer); longPressTimer = null; }
                longPressTriggered = false;
                pressStartPos = null;
            }
        }, { passive: false });

        canvas.addEventListener('touchmove', function(e) {
            if (isPinching && e.touches.length === 2) {
                e.preventDefault();
                var newDistance = getDistance(e.touches[0], e.touches[1]);
                var scale = newDistance / pinchStartDistance;
                var newCellSize = Math.round(pinchStartCellSize * scale);
                setCellSize(newCellSize, true);
            }
        }, { passive: false });

        canvas.addEventListener('touchend', function(e) {
            if (e.touches.length < 2) {
                isPinching = false;
                pinchStartDistance = 0;
            }
        });

        canvas.addEventListener('touchcancel', function() {
            isPinching = false;
            pinchStartDistance = 0;
        });

        canvas.addEventListener('pointerdown', onPointerDown);
        canvas.addEventListener('pointermove', onPointerMove);
        canvas.addEventListener('pointerup', onPointerUp);
        canvas.addEventListener('pointerleave', onPointerLeave);
        canvas.addEventListener('pointercancel', onPointerLeave);
        canvas.addEventListener('contextmenu', function(e) { e.preventDefault(); });

        btnRestart.addEventListener('click', restartGame);
        btnMenu.addEventListener('click', exitToMenu);
        smileyBtn.addEventListener('click', restartGame);
        btnHome.addEventListener('click', exitToMenu);
        btnAutoFlag.addEventListener('click', function() {
            autoFlagEnabled = !autoFlagEnabled;
            updateAutoFlagButton();
            if (game && !game.firstClick) restartGame();
        });

        btnCustomConfirm.addEventListener('click', function() {
            var rows = parseInt(inputRowsDialog.value) || 9;
            var cols = parseInt(inputColsDialog.value) || 9;
            var mines = parseInt(inputMinesDialog.value) || 10;
            rows = Math.max(5, Math.min(30, rows));
            cols = Math.max(5, Math.min(30, cols));
            var maxMines = rows * cols - 1;
            mines = Math.max(1, Math.min(maxMines, mines));
            customConfig = { rows: rows, cols: cols, mines: mines };
            customDialog.style.display = 'none';
            menuDiv.style.display = 'none';
            gameDiv.style.display = 'flex';
            initGame();
        });

        btnCustomCancel.addEventListener('click', function() { customDialog.style.display = 'none'; });

        document.addEventListener('keydown', function(e) {
            if (menuDiv.style.display !== 'none') {
                if (e.key === 'ArrowDown' || e.key === 'ArrowUp') {
                    e.preventDefault();
                    navigateMenu(e.key === 'ArrowDown' ? 1 : -1);
                } else if (e.key === 'Enter') {
                    e.preventDefault();
                    startGame(menuItems[menuIndex].dataset.mode);
                }
            } else if (e.key === 'Escape') {
                exitToMenu();
            }
        });

        menuItems.forEach(function(item, index) {
            item.addEventListener('click', function() {
                menuIndex = index;
                updateMenuSelection();
                startGame(item.dataset.mode);
            });
            item.addEventListener('mouseenter', function() {
                menuIndex = index;
                updateMenuSelection();
            });
        });

        document.addEventListener('touchmove', function(e) {
            if (gameDiv.style.display !== 'none' && !canvasWrapper.contains(e.target)) e.preventDefault();
        }, { passive: false });

        function handleResize() {
            updateZoomBounds();
            if (game && !game.gameOver && !game.win) restartGame();
        }
        window.addEventListener('resize', handleResize);

        function showNumKeyboard(inputField) { currentInputField = inputField; numKeyboard.style.display = 'grid'; }
        function hideNumKeyboard() { numKeyboard.style.display = 'none'; currentInputField = null; }

        inputRows.addEventListener('click', function() { showNumKeyboard(inputRows); });
        inputCols.addEventListener('click', function() { showNumKeyboard(inputCols); });
        inputMines.addEventListener('click', function() { showNumKeyboard(inputMines); });

        document.querySelectorAll('#num-keyboard button').forEach(function(btn) {
            btn.addEventListener('click', function() {
                if (!currentInputField) return;
                var key = btn.textContent;
                if (key === 'C') currentInputField.value = '';
                else if (key === '←') currentInputField.value = currentInputField.value.slice(0, -1);
                else if (key === '确定') {
                    hideNumKeyboard();
                    var rows = parseInt(inputRows.value) || 9;
                    var cols = parseInt(inputCols.value) || 9;
                    var mines = parseInt(inputMines.value) || 10;
                    rows = Math.max(5, Math.min(30, rows));
                    cols = Math.max(5, Math.min(30, cols));
                    var maxMines = rows * cols - 1;
                    mines = Math.max(1, Math.min(maxMines, mines));
                    inputRows.value = rows;
                    inputCols.value = cols;
                    inputMines.value = mines;
                    customConfig = { rows: rows, cols: cols, mines: mines };
                    restartGame();
                } else if (/^\d$/.test(key)) {
                    currentInputField.value += key;
                }
            });
        });

        updateMenuSelection();
    </script>
